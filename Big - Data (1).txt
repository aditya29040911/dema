counts = {}

    word, count = line.strip().split('\t')

for word, count in counts.items():

data = {}

for line in sys.stdin:
    gender, age = line.strip().split(',')

    if gender not in data:
        data[gender] = [0, 0]

    data[gender][0] += float(age)
    data[gender][1] += 1

for gender, values in data.items():

#!/usr/bin/env python
import sys

a, b = {}, {}

for line in sys.stdin:
    try:
        key, val = line.strip().split('\t', 1)
        p = val.split(',')
        tag = p[0].lower()
        i, j, v = int(p[1]), int(p[2]), int(p[3])

        if tag == 'a':
            a[i, j] = v
        elif tag == 'b':
            b[i, j] = v
    except:
        continue

for i in range(2):
    for j in range(2):
        s = sum(a.get((i, k), 0) * b.get((k, j), 0) for k in range(2))
        print "(%d,%d)\t%d" % (i, j, s)

4. Using PIG to find Most Frequent

[cloudera@quickstart ~]$ gedit word1.txt

1,pooja,R,20
2,sam,Thomas,25
3,priya,S,20
4,pavan,Rao,20

[cloudera@quickstart ~]$ pig -x local

grunt> A = LOAD '/home/cloudera/word1.txt' USING PigStorage(',') AS (id:int, first_name:chararray, last_name:chararray, age:int);

grunt> B = FOREACH A GENERATE UPPER(SUBSTRING(first_name, 0, 1)) AS first_letter;

grunt> C = GROUP B BY first_letter;

grunt> D = FOREACH C GENERATE group AS first_letter, COUNT(B) AS count;

grunt> E = ORDER D BY count DESC;

grunt> F = LIMIT E 1;

grunt> DUMP B;

grunt> DUMP D;

grunt> DUMP E;

grunt> DUMP F;

grunt> quit;

7. Retail Sales Analysis

[cloudera@quickstart ~]$ gedit retail.py

from pyspark import SparkContext
from pyspark.sql import SQLContext

sc = SparkContext()
sql = SQLContext(sc)

data = [
    ("Laptop", "Electronics", 50000),
    ("Phone", "Electronics", 30000),
    ("Rice", "Grocery", 5000),
    ("Oil", "Grocery", 2000),
    ("Chair", "Furniture", 4000),
    ("Table", "Furniture", 8000)
]

columns = ["Product", "Category", "Sales"]

df = sql.createDataFrame(data, columns)

print("All Products:")
df.show()

print("Products with sales above 5000:")
df.filter(df.Sales > 5000).show()

print("Total Sales by Category:")
df.groupBy("Category").sum("Sales").show()

print("Highest-Selling Product:")
df.orderBy(df.Sales.desc()).show(1)

print("Products sorted by Sales:")
df.orderBy(df.Sales.desc()).show()

sc.stop()
