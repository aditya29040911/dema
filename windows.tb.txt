import yfinance as yf
import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt
import statsmodels.api as sm


tickers=['INFY.NS','HINDUNILVR.NS','RELIANCE.NS','SBIN.NS']
market_ticker='^NSEI'
all_ticker=tickers+[market_ticker]

price=pd.DataFrame()
for t in all_ticker:
    price[t]=yf.download(t, start='2021-01-01', end='2026-08-01')['Close'].squeeze()

price.head()



all_return=price.pct_change().dropna() #(drop the first row as there is no prior data)
##infosys["return"]=(infosys["Close"]/infosys["Close"].shift(1))-1

daily_returns=all_return[tickers]
market_return=all_return[market_ticker]

daily_returns.head()


annual_returns=daily_returns.mean()*252
print('annualized expected return per stock:')
print((annual_returns*100).round(2).astype(str)+'%')



market_annual_returns=market_return.mean()*252
print(f'NIFTY_50 annualized return :{market_annual_returns*100:.2f}%')



weights= np.array([0.25,0.25,0.25,0.25]) # equal weights

portfolio_returns=np.dot(annual_returns,weights)
print(f'Portfolio expected return :{portfolio_returns*100:.2f}%')


cov_matrix=daily_returns.cov()*252
cov_matrix


portfolio_variance=weights@cov_matrix.values@weights
portfolio_volatility=portfolio_variance**0.5

print(f'Portfolio variance :{portfolio_variance:.4f}')
print(f'Portfolio volatility(risk) :{portfolio_volatility*100:.2f}%')


individual_vols=np.sqrt(np.diag(cov_matrix.values))
no_diversification_risk=np.dot(weights, individual_vols)

print(f'Portfolio risk(with diversificaton) :{portfolio_volatility*100:.2f}%')
print(f'risk if no diversification benifit :{no_diversification_risk*100:.2f}%')
print(f'diversification benifit :{(no_diversification_risk-portfolio_volatility)*100:.2f} percentage points')

corr_matrix=daily_returns.corr()
corr_matrix


plt.figure(figsize=(6,5))
sns.heatmap(corr_matrix, annot=True, cmap="coolwarm", vmin=-1, vmax=1,
           xticklabels=tickers, yticklabels=tickers)
plt.title("correlation matrix of daily return")
plt.tight_layout()
plt.show()     


RF_rate= 0.06

n_portfolios=5000
np.random.seed(42) #so result are producible

random_weights=np.random.dirichlet(np.ones(len(tickers)), n_portfolios)
port_returns=random_weights@annual_returns.values
port_risks=np.sqrt(np.einsum('ij,jk,ik->i',random_weights,cov_matrix.values,random_weights))
port_sharpes=(port_returns-RF_rate)/port_risks


max_sharpe_idx=np.argmax(port_sharpes)
min_risk_idx= np.argmin(port_risks)

print("Max Sharpe portfolio:")
print(f" Weights: {dict(zip(tickers, random_weights[max_sharpe_idx].round(3)))}") 
print(f" Return: {port_returns[max_sharpe_idx]*100:.2f}% Risk: {port_risks [max_sharpe_idx]*100:.2f}% Sharpe: {port_sharpes[max_sharpe_idx]:.3f}")

print("\nMin Risk (Minimum Variance) portfolio:")
print(f" Weights: {dict(zip(tickers, random_weights[min_risk_idx].round(3)))}")
print(f" Return: {port_returns [min_risk_idx]*100:.2f}% Risk: {port_risks [min_risk_idx]*100:.2f}%")



plt.figure(figsize=(9, 6))
plt.scatter(port_risks*100, port_returns*100, c=port_sharpes, cmap="viridis", s=8, alpha=0.6)
plt.colorbar(label="Sharpe Ratio")
plt.scatter(port_risks [max_sharpe_idx]*100, port_returns[max_sharpe_idx]*100,
            color="gold", marker="*", s=300, edgecolor="black", label="Max Sharpe Portfolio")
plt.scatter(port_risks [min_risk_idx]*100, port_returns[min_risk_idx]*100,
            color="green", marker="*", s=300, edgecolor="black", label="Min Risk Portfolio")
plt.xlabel("Risk (%)")
plt.ylabel("Expected Return (%)")
plt.title(f"Efficient Frontier -- {n_portfolios} Simulated Portfolios")
plt.legend()
plt.grid(True)
plt.show()



x= sm.add_constant(market_return.rename('Market')) # add the intercept

betas={}
alphas={}

for t in tickers:
    y=daily_returns[t]
    model=sm.OLS(y,x).fit()
    alphas[t]=model.params['const']
    betas[t]=model.params['Market']

betas=pd.Series(betas)
print("Beta of each stock(systematic risk vs nifty 50):")
print(betas.round(3))



portfolio_beta=np.dot(betas.values, weights)
print(f"protfolio beta:{portfolio_beta:.3f}")



def sharpe_ratio(port_return,port_volatility, rf=RF_rate):
    return(port_return-rf)/port_volatility

def treynor_ratio(port_return,port_beta, rf=RF_rate):
    return(port_return-rf)/port_beta

sharpe= sharpe_ratio(portfolio_returns, portfolio_volatility)
treynor= treynor_ratio(portfolio_returns, portfolio_beta)

print(f"Sharpe ratio: {sharpe:.4f}")
print(f"Treynor ratio: {treynor*100:.2f} % per unit of beta")



capm_predicted_return=RF_rate+portfolio_beta*(market_annual_returns-RF_rate)
jensons_alpha= portfolio_returns-capm_predicted_return

print(f"Market NIFTY 50 annualized return: {market_annual_returns*100:.2F} %")
print(f"portfolio beta: {portfolio_beta:.3f}")
print(f"CAPM predicted portfolio return: {capm_predicted_return*100:.2f} %")
print(f"Actual portfolio return: {portfolio_returns*100:.2f} %")
print(f"Jensons alpha: {jensons_alpha*100:.2f} %")

if jensons_alpha >0:
    print("\n--> positive alpha: the portfoio outperformed what capm predicts for its resk level.")
else:
    print("\n--> negative alpha: the portfoio underperformed what capm predicts for its resk level.")
